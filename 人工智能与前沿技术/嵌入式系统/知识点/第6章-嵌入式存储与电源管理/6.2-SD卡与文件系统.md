# 6.2 SD卡与文件系统

## 一、SD卡基础

### 1. SD卡类型

| 类型 | 容量 | 说明 |
|-----|------|------|
| SDSC | 最大2GB | 标准SD卡 |
| SDHC | 4GB~32GB | 高容量SD卡 |
| SDXC | 64GB~2TB | 扩展容量SD卡 |

### 2. SD卡接口

| 接口 | 说明 |
|-----|------|
| SPI模式 | 4线，简单 |
| SDIO模式 | 4位或1位模式，高速 |

### 3. SD卡SPI模式引脚

| 引脚 | 功能 |
|-----|------|
| CS (DAT3) | 片选 |
| SCK (CLK) | 时钟 |
| MOSI (CMD) | 数据输入（命令） |
| MISO (DAT0) | 数据输出 |
| VCC | 电源 (3.3V) |
| GND | 地 |

## 二、SD卡SPI模式驱动

### 1. SPI命令集

| 命令 | 代码 | 参数 | 说明 |
|-----|------|------|------|
| CMD0 | 0x40 | 0 | 复位 |
| CMD8 | 0x48 | 0x000001AA | 发送接口条件 |
| CMD55 | 0x77 | 0 | ACMD前缀 |
| ACMD41 | 0x69 | 0 | 发送操作条件 |
| CMD58 | 0x7A | 0 | 读取OCR寄存器 |
| CMD17 | 0x51 | 地址 | 读单块 |
| CMD24 | 0x58 | 地址 | 写单块 |
| CMD12 | 0x4C | 0 | 停止传输 |

### 2. SD卡初始化

```c
#define SD_CS_HIGH()  GPIO_SetBits(GPIOB, GPIO_Pin_12)
#define SD_CS_LOW()   GPIO_ResetBits(GPIOB, GPIO_Pin_12)

#define SD_CMD_GO_IDLE_STATE     0
#define SD_CMD_SEND_IF_COND      8
#define SD_CMD_SEND_OP_COND     41
#define SD_CMD_SEND_CSD          9
#define SD_CMD_SEND_CID         10
#define SD_CMD_SEND_DATA_START  17
#define SD_CMD_SET_BLOCKLEN     16

// 发送SD命令
uint8_t SD_SendCmd(uint8_t cmd, uint32_t arg, uint8_t crc) {
    uint8_t retry = 0;
    uint8_t res;
    
    SD_CS_HIGH();
    SPI_Transfer(0xFF);  // 时钟
    
    SD_CS_LOW();
    SPI_Transfer(cmd | 0x40);  // CMD + start bit
    SPI_Transfer((uint8_t)(arg >> 24));
    SPI_Transfer((uint8_t)(arg >> 16));
    SPI_Transfer((uint8_t)(arg >> 8));
    SPI_Transfer((uint8_t)(arg));
    SPI_Transfer(crc | 0x01);  // CRC + stop bit
    
    // 等待响应
    do {
        res = SPI_Transfer(0xFF);
        if(!(res & 0x80)) break;
        retry++;
    } while(retry < 100);
    
    return res;
}

// 初始化SD卡
uint8_t SD_Init(void) {
    uint8_t res;
    uint16_t retry = 0;
    uint8_t buf[4];
    
    // 上电初始化
    SD_CS_HIGH();
    for(int i = 0; i < 10; i++) {
        SPI_Transfer(0xFF);  // 发送80个时钟
    }
    
    // CMD0: 复位到空闲状态
    do {
        res = SD_SendCmd(SD_CMD_GO_IDLE_STATE, 0, 0x95);
        retry++;
    } while(res != 0x01 && retry < 200);
    
    if(retry >= 200) return 1;
    
    // CMD8: 发送接口条件（SD v2+）
    res = SD_SendCmd(SD_CMD_SEND_IF_COND, 0x000001AA, 0x87);
    if(res == 0x01) {
        // SD v2卡，读取响应
        for(int i = 0; i < 4; i++) {
            buf[i] = SPI_Transfer(0xFF);
        }
        if(buf[2] == 0x01 && buf[3] == 0xAA) {
            // 有效
        } else {
            return 2;
        }
    } else {
        // SD v1卡（忽略CMD8）
        retry = 0;
    }
    
    // ACMD41: 发送操作条件
    retry = 0;
    do {
        res = SD_SendCmd(SD_CMD_SEND_OP_COND, 0, 0xFF);
        retry++;
    } while(res != 0x00 && retry < 200);
    
    if(retry >= 200) return 3;
    
    // 读取OCR（CMD58）
    res = SD_SendCmd(58, 0, 0xFF);
    if(res != 0x00) return 4;
    
    for(int i = 0; i < 4; i++) {
        buf[i] = SPI_Transfer(0xFF);
    }
    
    // 判断SDHC
    if(buf[0] & 0x40) {
        sd_type = 2;  // SDHC
    } else {
        sd_type = 1;  // SDSC
    }
    
    // 设置块长度
    res = SD_SendCmd(SD_CMD_SET_BLOCKLEN, 512, 0xFF);
    if(res != 0x00) return 5;
    
    return 0;
}
```

### 3. SD卡数据读写

```c
// 读取一个块（512字节）
uint8_t SD_ReadBlock(uint32_t sector, uint8_t *buf) {
    uint16_t i;
    uint8_t res;
    uint8_t retry = 0;
    
    // SDSC: sector地址，SDHC: byte地址
    if(sd_type == 1) {
        sector <<= 9;  // *512
    }
    
    // 发送读取命令
    res = SD_SendCmd(SD_CMD_SEND_DATA_START, sector, 0xFF);
    if(res != 0x00) return 1;
    
    // 等待起始标志
    do {
        res = SPI_Transfer(0xFF);
        if(res == 0xFE) break;
        retry++;
    } while(retry < 100);
    
    if(retry >= 100) return 2;
    
    // 读取512字节
    for(i = 0; i < 512; i++) {
        buf[i] = SPI_Transfer(0xFF);
    }
    
    // 读取CRC（忽略）
    SPI_Transfer(0xFF);
    SPI_Transfer(0xFF);
    
    SD_CS_HIGH();
    
    return 0;
}

// 写入一个块（512字节）
uint8_t SD_WriteBlock(uint32_t sector, uint8_t *buf) {
    uint16_t i;
    uint8_t res;
    
    if(sd_type == 1) {
        sector <<= 9;
    }
    
    // 发送写命令
    res = SD_SendCmd(24, sector, 0xFF);
    if(res != 0x00) return 1;
    
    // 发送数据头
    SPI_Transfer(0xFF);  // 填充
    SPI_Transfer(0xFE);  // 起始标志
    
    // 发送512字节
    for(i = 0; i < 512; i++) {
        SPI_Transfer(buf[i]);
    }
    
    // 发送假CRC
    SPI_Transfer(0xFF);
    SPI_Transfer(0xFF);
    
    // 等待响应
    res = SPI_Transfer(0xFF);
    if((res & 0x1F) != 0x05) {
        return 2;
    }
    
    // 等待写入完成
    while(SPI_Transfer(0xFF) == 0x00);
    
    SD_CS_HIGH();
    
    return 0;
}
```

### 4. 磁盘IO接口（diskio.c）

```c
#include "diskio.h"
#include "SD_Card.h"

static DSTATUS sd_state = STA_NOINIT;

// 初始化磁盘
DSTATUS disk_initialize(BYTE pdrv) {
    if(pdrv != 0) return STA_NOINIT;
    
    if(SD_Init() == 0) {
        sd_state = 0;  // 初始化成功
    }
    
    return sd_state;
}

// 获取磁盘状态
DSTATUS disk_status(BYTE pdrv) {
    if(pdrv != 0) return STA_NOINIT;
    return sd_state;
}

// 读取扇区
DRESULT disk_read(BYTE pdrv, BYTE *buff, LBA_t sector, UINT count) {
    if(pdrv != 0 || !count) return RES_PARERR;
    if(sd_state == STA_NOINIT) return RES_NOTRDY;
    
    for(UINT i = 0; i < count; i++) {
        if(SD_ReadBlock(sector + i, buff + i * 512) != 0) {
            return RES_ERROR;
        }
    }
    
    return RES_OK;
}

// 写入扇区
DRESULT disk_write(BYTE pdrv, const BYTE *buff, LBA_t sector, UINT count) {
    if(pdrv != 0 || !count) return RES_PARERR;
    if(sd_state == STA_NOINIT) return RES_NOTRDY;
    
    for(UINT i = 0; i < count; i++) {
        if(SD_WriteBlock(sector + i, (BYTE*)buff + i * 512) != 0) {
            return RES_ERROR;
        }
    }
    
    return RES_OK;
}

// IO控制
DRESULT disk_ioctl(BYTE pdrv, BYTE cmd, void *buff) {
    if(pdrv != 0) return RES_PARERR;
    if(sd_state == STA_NOINIT) return RES_NOTRDY;
    
    switch(cmd) {
        case CTRL_SYNC:
            return RES_OK;
            
        case GET_SECTOR_COUNT:
            *(LBA_t*)buff = sd_sector_count;
            return RES_OK;
            
        case GET_SECTOR_SIZE:
            *(WORD*)buff = 512;
            return RES_OK;
            
        case GET_BLOCK_SIZE:
            *(DWORD*)buff = 1;
            return RES_OK;
            
        default:
            return RES_PARERR;
    }
}
```

### 5. FatFs使用示例

```c
#include "ff.h"

FATFS fs;      // 文件系统对象
FIL file;      // 文件对象
FRESULT res;   // 结果
UINT bw;       // 写入字节数

// 挂载文件系统
void fs_mount(void) {
    res = f_mount(&fs, "0:", 1);  // SD卡，立即挂载
    if(res != FR_OK) {
        printf("挂载失败: %d\n", res);
        return;
    }
    printf("文件系统已挂载\n");
}

// 创建文件并写入
void fs_write_file(const char *filename, const char *data) {
    res = f_open(&file, filename, FA_WRITE | FA_CREATE_ALWAYS);
    if(res != FR_OK) {
        printf("打开文件失败: %d\n", res);
        return;
    }
    
    res = f_write(&file, data, strlen(data), &bw);
    if(res != FR_OK || bw != strlen(data)) {
        printf("写入失败: %d\n", res);
    }
    
    f_close(&file);
}

// 读取文件
void fs_read_file(const char *filename, char *buf, UINT buf_size) {
    res = f_open(&file, filename, FA_READ);
    if(res != FR_OK) {
        printf("打开文件失败: %d\n", res);
        return;
    }
    
    res = f_read(&file, buf, buf_size, &bw);
    if(res != FR_OK) {
        printf("读取失败: %d\n", res);
    } else {
        buf[bw] = '\0';
        printf("读取到 %d 字节: %s\n", bw, buf);
    }
    
    f_close(&file);
}

// 格式化SD卡
void fs_format_sd(void) {
    BYTE work_buf[512];
    
    res = f_mkfs("0:", 0, work_buf, sizeof(work_buf));
    if(res != FR_OK) {
        printf("格式化失败: %d\n", res);
    } else {
        printf("格式化成功\n");
    }
    
    f_mount(NULL, "0:", 0);  // 卸载
    f_mount(&fs, "0:", 1);   // 重新挂载
}
```

### 1. 数据记录系统

```c
typedef struct {
    uint32_t timestamp;
    float temperature;
    float humidity;
    uint16_t pressure;
} SensorData;

// 数据日志
void data_log_task(void *pvParameters) {
    FIL log_file;
    SensorData sensor_data;
    FRESULT res;
    
    while(1) {
        // 采集传感器数据
        read_sensors(&sensor_data);
        
        // 打开日志文件（追加模式）
        res = f_open(&log_file, "0:/log.csv", 
                     FA_WRITE | FA_CREATE_ALWAYS | FA_OPEN_ALWAYS);
        
        if(res == FR_OK) {
            // 移到文件末尾
            f_lseek(&log_file, f_size(&log_file));
            
            // 写入CSV行
            char line[100];
            sprintf(line, "%lu,%.1f,%.1f,%u\n",
                    sensor_data.timestamp,
                    sensor_data.temperature,
                    sensor_data.humidity,
                    sensor_data.pressure);
            
            UINT bw;
            f_write(&log_file, line, strlen(line), &bw);
            f_close(&log_file);
        }
        
        vTaskDelay(pdMS_TO_TICKS(10000));  // 10秒记录一次
    }
}
```

## 五、注意事项

### 1. SD卡使用注意

| 问题 | 解决方案 |
|-----|---------|
| SPI模式速率 | 初始化时≤400KHz，正常≤25MHz |
| 块大小 | 固定512字节 |
| 读写超时 | 添加超时处理 |
| 热插拔 | 检测插入/移除 |

### 2. FatFs使用注意

| 问题 | 解决方案 |
|-----|---------|
| 栈空间 | 增加任务栈大小 |
| 缓冲区 | 使用FF_FS_TINY节省RAM |
| 并发访问 | 加互斥锁保护 |
| 错误恢复 | 检查返回值，处理错误 |
