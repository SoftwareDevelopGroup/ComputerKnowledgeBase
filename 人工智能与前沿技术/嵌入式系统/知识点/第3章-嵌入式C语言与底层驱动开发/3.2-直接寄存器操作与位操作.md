# 3.2 直接寄存器操作与位操作

## 一、寄存器地址映射

### 1. 获取寄存器地址

```c
// STM32F103 GPIOA基地址
#define GPIOA_BASE  0x40010800

// MODER寄存器偏移
#define GPIO_MODER  (*(volatile uint32_t*)(GPIOA_BASE + 0x00))
#define GPIO_OTYPER (*(volatile uint32_t*)(GPIOA_BASE + 0x04))
#define GPIO_OSPEEDR (*(volatile uint32_t*)(GPIOA_BASE + 0x08))
#define GPIO_PUPDR  (*(volatile uint32_t*)(GPIOA_BASE + 0x0C))
#define GPIO_IDR    (*(volatile uint32_t*)(GPIOA_BASE + 0x10))
#define GPIO_ODR    (*(volatile uint32_t*)(GPIOA_BASE + 0x14))
#define GPIO_BSRR   (*(volatile uint32_t*)(GPIOA_BASE + 0x18))
```

### 2. 使用指针访问

```c
// 使用结构体指针访问
typedef struct {
    volatile uint32_t MODER;
    volatile uint32_t OTYPER;
    volatile uint32_t OSPEEDR;
    volatile uint32_t PUPDR;
    volatile uint32_t IDR;
    volatile uint32_t ODR;
    volatile uint32_t BSRR;
    volatile uint32_t LCKR;
    volatile uint32_t AFRL;
    volatile uint32_t AFRH;
} GPIO_TypeDef;

#define GPIOA ((GPIO_TypeDef*)GPIOA_BASE)

// 访问GPIOA的MODER寄存器
void configure_gpio(void) {
    // 设置PA0为输出模式
    GPIOA->MODER &= ~(3 << 0);   // 清除bit[1:0]
    GPIOA->MODER |=  (1 << 0);   // 设置为输出模式(01)
    
    // 设置PA0的输出速度
    GPIOA->OSPEEDR &= ~(3 << 0);
    GPIOA->OSPEEDR |=  (2 << 0);  // 快速模式
    
    // 读取输入状态
    if(GPIOA->IDR & (1 << 0)) {
        // PA0为高电平
    }
    
    // 设置输出
    GPIOA->ODR |= (1 << 0);   // PA0输出高
    GPIOA->ODR &= ~(1 << 0);  // PA0输出低
}
```

## 二、位操作基础

### 1. 位操作符

| 操作符 | 说明 | 示例 |
|-------|------|------|
| `&` | 按位与 | `0xFF & 0x0F = 0x0F` |
| `\|` | 按位或 | `0xF0 \| 0x0F = 0xFF` |
| `^` | 按位异或 | `0xFF ^ 0x0F = 0xF0` |
| `~` | 按位取反 | `~0x01 = 0xFE` (8位) |
| `<<` | 左移 | `1 << 3 = 8` |
| `>>` | 右移 | `8 >> 2 = 2` |

### 2. 常用位操作宏

```c
// 位操作宏
#define SET_BIT(REG, BIT)    ((REG) |= (BIT))
#define CLEAR_BIT(REG, BIT)  ((REG) &= ~(BIT))
#define TOGGLE_BIT(REG, BIT) ((REG) ^= (BIT))
#define READ_BIT(REG, BIT)   ((REG) & (BIT))
#define WRITE_BIT(REG, BIT, VAL)  do { if(VAL) SET_BIT(REG, BIT); else CLEAR_BIT(REG, BIT); } while(0)

// 字节操作
#define SET_BYTE(REG, VAL)   ((REG) = (VAL))
#define GET_BYTE(REG)        ((REG) & 0xFF)

// 位域操作
#define SET_FIELD(REG, MASK, VAL)  ((REG) = ((REG) & ~(MASK)) | ((VAL) & (MASK)))
#define GET_FIELD(REG, MASK)       ((REG) & (MASK))
```

### 3. 位操作示例

```c
// 示例：配置GPIOA的PA5为输出
void GPIOA_PA5_Output_Init(void) {
    // 1. 使能时钟
    RCC->AHBENR |= RCC_AHBENR_GPIOAEN;
    
    // 2. 配置模式：输出
    // MODER[11:10] = 01 (输出模式)
    GPIOA->MODER &= ~(3 << 10);  // 清除bit[11:10]
    GPIOA->MODER |=  (1 << 10);  // 设置为输出
    
    // 3. 配置输出速度
    // OSPEEDR[11:10] = 10 (快速)
    GPIOA->OSPEEDR &= ~(3 << 10);
    GPIOA->OSPEEDR |=  (2 << 10);
    
    // 4. 配置输出类型：推挽
    // OTYPER[5] = 0 (推挽)
    GPIOA->OTYPER &= ~(1 << 5);
    
    // 5. 配置上下拉：无
    // PUPDR[11:10] = 00 (无上下拉)
    GPIOA->PUPDR &= ~(3 << 10);
    
    // 6. 设置初始电平
    GPIOA->BSRR = (1 << 21);  // 复位ODR5 (ODR置0)
    // 或
    GPIOA->BSRR = (1 << 5);   // 设置ODR5 (ODR置1)
}
```

## 三、读-改-写操作模式

### 1. 什么是读-改-写

> [!important] 读-改-写（Read-Modify-Write）
> 对寄存器的部分位进行修改时，需要先读取整个寄存器，修改目标位，再写回。

```
1. 读：读取寄存器当前值
2. 改：修改需要修改的位
3. 写：写回修改后的值
```

### 2. 正确的读-改-写

```c
// 示例：只修改GPIOA->MODER的bit[11:10]，其他位不变
void set_pa5_output(void) {
    // 1. 读取当前值
    uint32_t moder = GPIOA->MODER;
    
    // 2. 清除目标位
    moder &= ~(3 << 10);  // 清除bit[11:10]
    
    // 3. 设置新值
    moder |= (1 << 10);   // 设置输出模式
    
    // 4. 写回
    GPIOA->MODER = moder;
}
```

### 3. 错误的写法

```c
// 错误！直接赋值会覆盖所有位
void set_pa5_output_wrong(void) {
    GPIOA->MODER = (1 << 10);  // 错误！所有其他位都被清零
}
```

### 4. BSRR寄存器的优势

```c
// BSRR寄存器：直接设置/清除指定位，无需读-改-写
// 低16位：设置ODR对应位
// 高16位：清除ODR对应位

// 设置PA5为高电平
GPIOA->BSRR = (1 << 5);    // 原子操作，不影响其他位

// 设置PA5为低电平
GPIOA->BSRR = (1 << 21);   // 16+5 = 21

// 同时设置多个位
GPIOA->BSRR = (1 << 5) | (1 << 6);

// 原子切换（推荐）
void toggle_pa5(void) {
    if(GPIOA->ODR & (1 << 5)) {
        GPIOA->BSRR = (1 << 21);  // 清除
    } else {
        GPIOA->BSRR = (1 << 5);   // 设置
    }
}
```

## 四、位域（Bit-Field）

### 1. 位域定义

```c
// 使用位域定义寄存器结构
typedef union {
    struct {
        uint32_t MODER0   : 2;  // PA0模式
        uint32_t MODER1   : 2;  // PA1模式
        uint32_t MODER2   : 2;  // PA2模式
        uint32_t MODER3   : 2;  // PA3模式
        uint32_t MODER4   : 2;  // PA4模式
        uint32_t MODER5   : 2;  // PA5模式
        uint32_t MODER6   : 2;  // PA6模式
        uint32_t MODER7   : 2;  // PA7模式
        uint32_t MODER8   : 2;  // PA8模式
        uint32_t MODER9   : 2;  // PA9模式
        uint32_t MODER10  : 2;  // PA10模式
        uint32_t MODER11  : 2;  // PA11模式
        uint32_t MODER12  : 2;  // PA12模式
        uint32_t MODER13  : 2;  // PA13模式
        uint32_t MODER14  : 2;  // PA14模式
        uint32_t MODER15  : 2;  // PA15模式
    } bit;
    uint32_t all;
} GPIO_MODER_TypeDef;
```

### 2. 位域使用

```c
// 使用位域配置寄存器
void configure_gpio_with_bitfield(void) {
    GPIO_MODER_TypeDef moder;
    moder.all = GPIOA->MODER;
    
    moder.bit.MODER5 = 1;  // PA5设为输出模式
    
    GPIOA->MODER = moder.all;
}
```

### 3. 位域的优缺点

| 优点 | 缺点 |
|-----|------|
| 代码清晰 | 编译器相关，可移植性差 |
| 可按位访问 | 可能影响性能 |
| 与结构体结合 | 不适合原子操作 |

## 五、完整的寄存器级GPIO驱动

```c
// gpio_driver.h
#ifndef GPIO_DRIVER_H
#define GPIO_DRIVER_H

#include "stm32f10x.h"

// GPIO模式枚举
typedef enum {
    GPIO_MODE_INPUT = 0,
    GPIO_MODE_OUTPUT = 1,
    GPIO_MODE_AF = 2,
    GPIO_MODE_ANALOG = 3
} GPIO_Mode_t;

// GPIO输出类型枚举
typedef enum {
    GPIO_TYPE_PUSHPULL = 0,
    GPIO_TYPE_OPENDRAIN = 1
} GPIO_OutputType_t;

// GPIO速度枚举
typedef enum {
    GPIO_SPEED_LOW = 0,
    GPIO_SPEED_MEDIUM = 1,
    GPIO_SPEED_FAST = 2,
    GPIO_SPEED_VERYFAST = 3
} GPIO_Speed_t;

// GPIO配置结构体
typedef struct {
    uint8_t pin;           // 引脚号 (0~15)
    GPIO_Mode_t mode;      // 模式
    GPIO_OutputType_t otype; // 输出类型
    GPIO_Speed_t speed;    // 速度
    uint8_t pull;          // 上下拉 (0=无, 1=上拉, 2=下拉)
} GPIO_Config_t;

// 函数声明
void GPIO_Init(GPIO_TypeDef *GPIOx, GPIO_Config_t *config);
void GPIO_WritePin(GPIO_TypeDef *GPIOx, uint8_t pin, uint8_t state);
uint8_t GPIO_ReadPin(GPIO_TypeDef *GPIOx, uint8_t pin);
void GPIO_TogglePin(GPIO_TypeDef *GPIOx, uint8_t pin);

#endif
```

```c
// gpio_driver.c
#include "gpio_driver.h"

void GPIO_Init(GPIO_TypeDef *GPIOx, GPIO_Config_t *config) {
    uint32_t moder_mask = 3 << (config->pin * 2);
    uint32_t speed_mask = 3 << (config->pin * 2);
    uint32_t pupdr_mask = 3 << (config->pin * 2);
    
    // 1. 配置模式
    GPIOx->MODER &= ~moder_mask;
    GPIOx->MODER |= ((uint32_t)config->mode << (config->pin * 2));
    
    // 2. 配置输出类型
    if(config->mode == GPIO_MODE_OUTPUT || config->mode == GPIO_MODE_AF) {
        if(config->otype == GPIO_TYPE_OPENDRAIN) {
            GPIOx->OTYPER |= (1 << config->pin);
        } else {
            GPIOx->OTYPER &= ~(1 << config->pin);
        }
    }
    
    // 3. 配置速度
    GPIOx->OSPEEDR &= ~speed_mask;
    GPIOx->OSPEEDR |= ((uint32_t)config->speed << (config->pin * 2));
    
    // 4. 配置上下拉
    GPIOx->PUPDR &= ~pupdr_mask;
    GPIOx->PUPDR |= ((uint32_t)config->pull << (config->pin * 2));
}

void GPIO_WritePin(GPIO_TypeDef *GPIOx, uint8_t pin, uint8_t state) {
    if(state) {
        GPIOx->BSRR = (1 << pin);       // 设置ODR
    } else {
        GPIOx->BSRR = (1 << (pin + 16)); // 清除ODR
    }
}

uint8_t GPIO_ReadPin(GPIO_TypeDef *GPIOx, uint8_t pin) {
    if(GPIOx->IDR & (1 << pin)) {
        return 1;
    }
    return 0;
}

void GPIO_TogglePin(GPIO_TypeDef *GPIOx, uint8_t pin) {
    if(GPIO_ReadPin(GPIOx, pin)) {
        GPIO_WritePin(GPIOx, pin, 0);
    } else {
        GPIO_WritePin(GPIOx, pin, 1);
    }
}
```
