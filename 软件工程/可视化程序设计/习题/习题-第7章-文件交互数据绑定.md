<details>
<summary>四、简答题答案</summary>

**7.21** `FileChooser` 只负责选路径（弹原生对话框返回 `File`），实际读写交给 Java NIO `Files`（`readString`/`writeString`/`readAllLines`）。`showSaveDialog` 不保证扩展名是因为它只产路径，用户输入的文件名可能不带扩展名，且文件可能不存在需 NIO 写入时创建，故扩展名需手动补齐。

**7.22** `ObservableList` 是带监听器的列表，增删元素触发 `ListChangeListener`，`ListView`/`TableView` 内部注册了监听器，数据变化时自动重绘。用普通 `ArrayList` 替代则不触发监听，调用 `add` 后界面不刷新，需手动 `setItems` 重建。

**7.23** `Preferences`：少量键值，不可见，不跨语言，用于窗口位置/最近文件/勾选项。Java 序列化：中等数据量，二进制不可读，不跨语言，用于 Java 内部对象图缓存，对版本敏感。JSON（Gson/Jackson）：中大数据量，文本可读，跨语言，用于配置/用户数据/数据交换，易版本演进。

**7.24** `PropertyValueFactory("name")` 通过反射优先查找 `nameProperty()` 方法（返回 `Property`），若不存在降级用 `getName()`。模型类应暴露 `nameProperty()`，这样单元格能监听属性变化实现即时更新（如编辑回写、外部修改同步）；仅用 `getXxx()` 则单元格无法监听后续变化。
</details>

```java
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import java.nio.file.*;

public class ConfigApp extends Application {
    public static class AppConfig {
        String theme = "light";
        int fontSize = 14;
        boolean autoSave = true;
        String lastFile = "";
        @Override public String toString() {
            return "theme=" + theme + ", fontSize=" + fontSize
                    + ", autoSave=" + autoSave + ", lastFile=" + lastFile;
        }
    }

    private final Gson gson = new GsonBuilder().setPrettyPrinting().create();
    private final Path configPath = Path.of("config.json");
    private AppConfig config = new AppConfig();

    @Override
    public void start(Stage stage) {
        Label show = new Label(config.toString());
        Button saveBtn = new Button("保存配置");
        Button loadBtn = new Button("加载配置");

        saveBtn.setOnAction(e -> {
            try {
                Files.writeString(configPath, gson.toJson(config));
                show.setText("已保存: " + config);
            } catch (Exception ex) {
                show.setText("保存失败: " + ex.getMessage());
            }
        });
        loadBtn.setOnAction(e -> {
            try {
                if (Files.exists(configPath)) {
                    config = gson.fromJson(Files.readString(configPath), AppConfig.class);
                    show.setText("已加载: " + config);
                } else {
                    show.setText("配置文件不存在");
                }
            } catch (Exception ex) {
                show.setText("加载失败: " + ex.getMessage());
            }
        });

        VBox root = new VBox(10, saveBtn, loadBtn, show);
        root.setStyle("-fx-padding: 20;");
        stage.setScene(new Scene(root, 360, 180));
        stage.setTitle("JSON 配置读写");
        stage.show();
    }
    public static void main(String[] args) { launch(args); }
}
```

**7.30** 参考代码（带记忆的记事本，Gson 部分注释说明）：

```java
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.BorderPane;
import javafx.stage.Stage;
import java.io.File;
import java.nio.file.*;
import java.nio.charset.StandardCharsets;
import java.util.*;
import java.util.prefs.Preferences;
// import com.google.gson.Gson;
// import com.google.gson.reflect.TypeToken;

public class MemoryNotepad extends Application {
    private final TextArea area = new TextArea();
    private final FileChooser chooser = new FileChooser();
    private final Preferences prefs = Preferences.userNodeForPackage(MemoryNotepad.class);
    // private final Gson gson = new Gson();
    private final List<String> recentFiles = new ArrayList<>();
    private Stage stage;

    @Override
    public void start(Stage stage) {
        this.stage = stage;
        chooser.getExtensionFilters().add(
                new FileChooser.ExtensionFilter("文本文件", "*.txt"));

        // 恢复窗口状态
        stage.setX(prefs.getDouble("x", 100));
        stage.setY(prefs.getDouble("y", 100));
        stage.setWidth(prefs.getDouble("w", 700));
        stage.setHeight(prefs.getDouble("h", 500));

        // 加载最近文件（JSON 简化为 Preferences 存路径列表）
        loadRecent();

        Menu fileMenu = new Menu("文件");
        MenuItem open = new MenuItem("打开");
        MenuItem save = new MenuItem("保存");
        MenuItem exit = new MenuItem("退出");
        open.setOnAction(e -> openFile());
        save.setOnAction(e -> saveFile());
        exit.setOnAction(e -> { saveWindowState(); stage.close(); });
        fileMenu.getItems().addAll(open, save, new SeparatorMenuItem(),
                buildRecentMenu(), new SeparatorMenuItem(), exit);

        BorderPane root = new BorderPane();
        root.setTop(new MenuBar(fileMenu));
        root.setCenter(area);

        stage.setOnCloseRequest(e -> { e.consume(); saveWindowState(); stage.close(); });
        stage.setScene(new Scene(root, 700, 500));
        stage.setTitle("带记忆的记事本");
        stage.show();
    }

    private Menu buildRecentMenu() {
        Menu recent = new Menu("最近文件");
        for (String p : recentFiles) {
            MenuItem item = new MenuItem(p);
            item.setOnAction(e -> {
                try {
                    area.setText(Files.readString(Path.of(p), StandardCharsets.UTF_8));
                } catch (Exception ex) { /* 忽略 */ }
            });
            recent.getItems().add(item);
        }
        return recent;
    }

    private void loadRecent() {
        // 实际项目用 Gson 从 recent.json 读取，这里简化用 Preferences
        for (int i = 0; i < 5; i++) {
            String p = prefs.get("recent" + i, null);
            if (p != null) recentFiles.add(p);
        }
    }

    private void addRecent(String path) {
        recentFiles.remove(path);
        recentFiles.add(0, path);
        if (recentFiles.size() > 5) recentFiles.remove(recentFiles.size() - 1);
        for (int i = 0; i < recentFiles.size(); i++) {
            prefs.put("recent" + i, recentFiles.get(i));
        }
    }

    private void openFile() {
        File f = chooser.showOpenDialog(stage);
        if (f == null) return;
        try {
            area.setText(Files.readString(f.toPath(), StandardCharsets.UTF_8));
            addRecent(f.getAbsolutePath());
            stage.setTitle(f.getName() + " - 记事本");
        } catch (Exception ex) {
            new Alert(Alert.AlertType.ERROR, "打开失败").show();
        }
    }

    private void saveFile() {
        File f = chooser.showSaveDialog(stage);
        if (f == null) return;
        try {
            Path p = f.toPath();
            if (!p.toString().endsWith(".txt")) p = Path.of(p + ".txt");
            Files.writeString(p, area.getText(), StandardCharsets.UTF_8);
            addRecent(p.toString());
            stage.setTitle(p.getFileName() + " - 记事本");
        } catch (Exception ex) {
            new Alert(Alert.AlertType.ERROR, "保存失败").show();
        }
    }

    private void saveWindowState() {
        prefs.putDouble("x", stage.getX());
        prefs.putDouble("y", stage.getY());
        prefs.putDouble("w", stage.getWidth());
        prefs.putDouble("h", stage.getHeight());
    }
```
